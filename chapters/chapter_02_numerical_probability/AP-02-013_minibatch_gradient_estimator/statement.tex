        \subsection{Mathematical statement}


Given per-example gradients $\bm g_i\in\mathbb R^d$, sample a uniform subset
$B\subset\{1,\ldots,N\}$ of size $b$ without replacement and define
\[
\hat{\bm g}_B=\frac1b\sum_{i\in B}\bm g_i,
\qquad E[\hat{\bm g}_B]=\bar{\bm g}.
\]
For coordinate $j$ with finite-population sample variance $S_j^2$,
\[
\operatorname{Var}(\hat g_{B,j})=\left(1-\frac bN\right)\frac{S_j^2}{b}.
\]

        \subsection{Theory--engineering gap inventory}

        \begin{gapbox}
        \textbf{Explicit gap.} An unbiased estimator is a distributional statement, not equality for one minibatch; software additionally chooses replacement policy, RNG state, repetitions, and diagnostics.

        \textbf{Implicit gap exposed by the result.} Empirical variance shrinks with batch size and vanishes at $b=N$ according to the finite-population correction, which an independent-with-replacement mental model misses.
        \end{gapbox}

        \subsection{Acceptance evidence}

        \begin{evidencebox}
        \begin{itemize}
          \item Same-seed samples are identical; a different seed changes them.
  \item Coordinate mean errors are reported in standard-error units and remain within the stated five-sigma test envelope for a fixed deterministic experiment.
  \item The benchmark plots empirical/theoretical trace variance over batch size and prints their ratios. (Trust-based: the test suite does not machine-check benchmark output.)
        \end{itemize}
        \end{evidencebox}

        \subsection{Constraints}

        \begin{itemize}
          \item Sample each repeat without replacement via a local \code{Generator}.
  \item Do not enumerate all subsets and do not use a framework data loader.
  \item Reject $b>N$, nonpositive repeats, nonfinite values, or non-matrix input with \code{ValueError}.
        \end{itemize}
