        \subsection{Mathematical statement}


For probability vectors $\bm p,\bm q\in\Delta^{K-1}$,
\[
\begin{gathered}
H(\bm p)=-\sum_k p_k\log p_k,\\
D_{\mathrm{KL}}(\bm p\|\bm q)=\sum_k p_k\log\frac{p_k}{q_k}.
\end{gathered}
\]
Use the measure-theoretic boundary conventions
\[
0\log0=0,\qquad 0\log(0/q)=0,\qquad p\log(p/0)=+\infty\;(p>0).
\]

        \subsection{Theory--engineering gap inventory}

        \begin{gapbox}
        \textbf{Explicit gap.} Interior calculus formulas omit extended-real boundary values and the validation needed to establish that arrays are probability measures.

        \textbf{Implicit gap exposed by the result.} Zeros shared by $p$ and $q$ should contribute exactly zero, while support mismatch must produce infinity rather than NaN or a clipped finite number.
        \end{gapbox}
