\problemstart{AP-02-003}{Softmax With Shift Invariance and a Named Axis}{Difficulty 3/5}{Chapter 2 \textbullet\ numpy}{Compute normalized class probabilities for extreme logits while preserving axis and shift invariants.}

        \subsection{Mathematical statement}


For logits $Z\in\mathbb R^{N\times K}$,
\[
  P_{ik}=\frac{e^{Z_{ik}-m_i}}{\sum_{j=1}^{K}e^{Z_{ij}-m_i}},
  \qquad m_i=\max_jZ_{ij}.
\]
For any row-wise shift $a_i$,
\[
  \operatorname{softmax}(Z+\bm a\bm 1_K^\top)
  =\operatorname{softmax}(Z),\qquad
  P\bm 1_K=\bm 1_N.
\]

        \subsection{Theory--engineering gap inventory}

        \begin{gapbox}
        \textbf{Explicit gap.} The ratio of exponentials is well-defined over reals, but a program must control overflow, reduction axis, retained dimensions, and summation rounding.

        \textbf{Implicit gap exposed by the result.} Adding $10^4$ to every logit leaves the mathematics unchanged but causes a naive implementation to produce non-finite probabilities.
        \end{gapbox}

        \subsection{Acceptance evidence}

        \begin{evidencebox}
        \begin{itemize}
          \item Extreme logits yield finite nonnegative probabilities whose selected-axis sums are one.
  \item Shift-invariance error and SciPy-oracle error are measured independently.
  \item A non-last-axis fixture proves the axis argument is not decorative.
        \end{itemize}
        \end{evidencebox}

        \subsection{Constraints}

        \begin{itemize}
          \item Use NumPy array primitives; do not call SciPy softmax in \code{starter.py}.
  \item Do not clip logits or add epsilon to the denominator.
  \item Preserve the input floating dtype.
        \end{itemize}
